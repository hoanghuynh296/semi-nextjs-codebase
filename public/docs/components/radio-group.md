# Radio group

Exactly one option of a few, each with an optional description; vertical by default.

## When to use

- 2–5 options where the descriptions help the choice (plans, shipping).

## Use something else when

| Need | Use |
|---|---|
| A filter on a list | Pills |
| More than 5 options | CustomSelect |

## Import

```tsx
import { RadioGroup } from "@semi/ui";
```

## Examples

### Plans

```tsx
"use client";

import { useState } from "react";
import { RadioGroup } from "@semi/ui";

export default function RadioGroupBasic() {
  const [plan, setPlan] = useState<"free" | "pro" | "team">("pro");
  return (
    <RadioGroup
      label="Gói dịch vụ"
      value={plan}
      onChange={setPlan}
      options={[
        { value: "free", label: "Free", description: "1 máy in" },
        { value: "pro", label: "Pro", description: "Tối đa 10 máy" },
        { value: "team", label: "Team", description: "Không giới hạn", disabled: true },
      ]}
    />
  );
}
```
