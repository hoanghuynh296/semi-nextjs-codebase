# Chip

A pill-shaped toggle for filters and tags. Neutral when off; active uses the high-contrast surface, with colour only on its dot or icon (tone).

## When to use

- Several filters that can be on at once (materials, device types).
- Removable active filters above a list.

## Use something else when

| Need | Use |
|---|---|
| Exactly one of 2–5 options | Pills |
| A read-only status | Badge |

## Import

```tsx
import { Chip } from "@semi/ui";
```

## Examples

### Toggle filters

tone colours the dot of what the chip filters; the chip itself stays neutral.

```tsx
"use client";

import { useState } from "react";
import { Chip } from "@semi/ui";

const MATERIALS = [
  { key: "pla", label: "PLA", tone: "green" },
  { key: "petg", label: "PETG", tone: "sky" },
  { key: "tpu", label: "TPU", tone: "pink" },
] as const;

export default function ChipToggle() {
  const [on, setOn] = useState<string[]>(["pla"]);
  const toggle = (k: string) => setOn((v) => (v.includes(k) ? v.filter((x) => x !== k) : [...v, k]));
  return (
    <div className="flex flex-wrap gap-2">
      {MATERIALS.map((m) => (
        <Chip key={m.key} tone={m.tone} active={on.includes(m.key)} onClick={() => toggle(m.key)}>
          {m.label}
        </Chip>
      ))}
    </div>
  );
}
```
